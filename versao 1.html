let estado = {

  usuario: { idade: 17, temTitulo: 'sim' },

  respostas: [], // Otimização: Armazenar respostas para analytics real

  cenarioAtual: 0,

  timerInterval: null,

  tempoRestante: 40, // 4.0s (em decibéis de segundo para animação fluida)

  tempoTotalLeitura: 0

};


function startExperiment() {

  estado.usuario.idade = parseInt(document.getElementById('input-idade').value) || 17;

  estado.usuario.temTitulo = document.getElementById('input-titulo').value;

  

  document.getElementById('screen-intro').classList.add('hidden');

  document.getElementById('screen-game').classList.remove('hidden');

  

  carregarCenario();

}


function carregarCenario() {

  if (estado.cenarioAtual >= cenarios.length) {

    finalizarExperimento();

    return;

  }

  

  const cenario = cenarios[estado.cenarioAtual];

  document.getElementById('game-progress').innerText = `CENÁRIO ${estado.cenarioAtual + 1} DE ${cenarios.length}`;

  document.getElementById('scenario-title').innerText = cenario.titulo;

  document.getElementById('btn-optA').innerText = cenario.optA;

  document.getElementById('btn-optB').innerText = cenario.optB;

  

  iniciarTimer();

}


function iniciarTimer() {

  clearInterval(estado.timerInterval);

  estado.tempoRestante = 40;

  atualizarUiTimer();

  

  estado.timerInterval = setInterval(() => {

    estado.tempoRestante--;

    atualizarUiTimer();

    

    if (estado.tempoRestante <= 0) {

      submitAnswer('TIMEOUT');

    }

  }, 100);

}


function atualizarUiTimer() {

  const segundosFormatados = Math.max(0, (estado.tempoRestante / 10)).toFixed(1);

  document.getElementById('game-timer').innerText = `${segundosFormatados.padStart(4, '0')}s`;

  document.getElementById('timer-fill').style.width = `${(Math.max(0, estado.tempoRestante) / 40) * 100}%`;

}


function submitAnswer(opcaoSelecionada) {

  clearInterval(estado.timerInterval); // Previne memory leak e execuções duplas

  

  // Otimização: Salvar a resposta no estado

  estado.respostas.push({

    cenarioId: cenarios[estado.cenarioAtual].id,

    resposta: opcaoSelecionada,

    tempoGastoSegundos: ((40 - Math.max(0, estado.tempoRestante)) / 10).toFixed(1)

  });

  

  console.log("Log de Analytics Parcial:", estado.respostas); // Útil para debugar

  

  estado.cenarioAtual++;

  carregarCenario();

}

function finalizarExperimento() {

  document.getElementById('screen-game').classList.add('hidden');

  document.getElementById('screen-results').classList.remove('hidden');

  

  renderizarEstatistica();

  renderizarAbas();

}


function renderizarEstatistica() {

  const banner = document.getElementById('stats-banner');

  const porcentagem = estado.usuario.temTitulo === 'sim' ? "+38%" : "-29%";

  const textoStatus = estado.usuario.temTitulo === 'sim' ? 'JÁ possuem' : 'NÃO possuem';

  

  banner.innerHTML = `

    <span style="color: var(--muted); font-size: 0.8rem; letter-spacing: 1px;">MÉTRICA DO SEU GRUPO (${estado.usuario.idade} ANOS):</span><br><br>

    Usuários que <strong>${textoStatus}</strong> título de eleitor dedicam, em média, 

    <span style="color: var(--rust); font-weight:bold;">${porcentagem} de tempo</span> à análise fria de dados antes de compartilhar um resultado online.

  `;

}


function renderizarAbas() {

  const tabsContainer = document.getElementById('tabs-container');

  tabsContainer.innerHTML = '';

  

  const chaves = Object.keys(dadosMacro);

  

  chaves.forEach((key, index) => {

    const btn = document.createElement('button');

    btn.className = `tab-btn ${index === 0 ? 'active' : ''}`;

    btn.innerText = dadosMacro[key].label;

    btn.onclick = (e) => alternarAba(key, e.target);

    tabsContainer.appendChild(btn);

  });

  

  // Força a renderização da primeira aba

  alternarAba(chaves[0], tabsContainer.firstChild);

}


function alternarAba(chaveId, elementoBotao) {

  document.querySelectorAll('.tab-btn').forEach(b => b.classList.remove('active'));

  elementoBotao.classList.add('active');

  

  const dados = dadosMacro[chaveId];

  

  document.getElementById('tab-content').innerHTML = `

      <article class="card">

        <h3>${dados.abordagemA.titulo}</h3>
